def invert_extrinsics(extrinsic_matrices):
    n = extrinsic_matrices.shape[0]
    inverted_list = []

    for i in range(n):
        M = extrinsic_matrices[i]
        R = M[:3, :3]
        t = M[:3, 3]

        # Calculate inverse
        R_inv = R.T
        t_inv = -R_inv @ t

        # Reconstruct
        M_inv = np.eye(4)
        M_inv[:3, :3] = R_inv
        M_inv[:3, 3] = t_inv
        
        inverted_list.append(M_inv)

    return np.array(inverted_list)
